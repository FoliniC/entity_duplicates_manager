def ensure_dir(path: str):
    """Ensure the directory for the given file or directory path exists."""
    if os.path.splitext(path)[1]:
        os.makedirs(os.path.dirname(path), exist_ok=True)
    else:
        os.makedirs(path, exist_ok=True)


def load_fancoil_vars(config_path):
    """Load room variables from YAML."""
    logger.info(f"Loading vars from {config_path}")
    try:
        with open(config_path, "r", encoding="utf-8") as f:
            config = yaml.safe_load(f) or {}
    except Exception as e:
        logger.error(f"Error loading YAML config: {e}")
        raise

    if not isinstance(config, dict):
        raise ValueError("Top-level YAML must be a mapping")

    if "fancoil_vars" in config and isinstance(config["fancoil_vars"], dict):
        vars_map = config["fancoil_vars"]
    else:
        vars_map = config

    for k, v in vars_map.items():
        if not isinstance(v, dict):
            raise ValueError(f"Item '{k}' must be a mapping, got {type(v)}")

    logger.info(f"Loaded fancoil_vars data: {vars_map}")
    logger.info("Successfully loaded vars")
    return vars_map


def substitute_tags(template_text: str, context: dict) -> str:
    """Replace every %%key%% with its context value. Signals error if spaces are found."""
    def replace_match(match):
        raw_tag = match.group(0)
        inner = match.group(1)
        key = inner.strip()
        
        if inner != key:
            logger.error(f"Malformed tag found with spaces: '{raw_tag}'. "
                         f"Spaces are NOT allowed inside %%. Use %%{key}%%")
        
        if key in context:
            return str(context[key])
        return raw_tag

    # Regex matching %%any_content%%
    return re.sub(r"%%(.*?)%%", replace_match, template_text)


def resolve_output_name(context: dict, default_key: str, suffix: str) -> str:
    """Use context['filename'] as base name; otherwise default_key. Append suffix and '.yaml'."""
    raw = str(context.get("filename", default_key)).strip()
    raw = os.path.basename(raw)
    base, _ext = os.path.splitext(raw)
    if not base:
        base = default_key
    base = re.sub(r"[^A-Za-z0-9_.-]+", "_", base)
    return f"{base}{suffix}.yaml"


def read_template(path: str) -> str:
    try:
        with open(path, "r", encoding="utf-8") as f:
            return f.read()
    except UnicodeDecodeError:
        # Try with latin-1 if utf-8 fails
        try:
            with open(path, "r", encoding="latin-1") as f:
                return f.read()
        except Exception as e:
            logger.error(f"Error reading template file {path}: {e}")
            raise
    except Exception as e:
        logger.error(f"Error reading template file {path}: {e}")
        raise


def clean_yaml_content(text: str) -> str:
    """Remove leading spaces from each line and trim trailing whitespace."""
    lines = text.split("\n")
    cleaned_lines = []
    for line in lines:
        # Remove trailing whitespace
        line = line.rstrip()
        # Don't remove all leading spaces - preserve YAML indentation
        # Just ensure no extra spaces before the YAML structure
        if line and not line.startswith(" "):
            # This is a top-level line, ensure it starts at column 0
            line = line.lstrip()
        cleaned_lines.append(line)
    return "\n".join(cleaned_lines)


def write_text_file(path: str, text: str):
    ensure_dir(path)
    try:
        # Clean the content before writing
        cleaned_text = clean_yaml_content(text)
        with open(path, "w", encoding="utf-8", newline="\n") as of:
            of.write(cleaned_text)
        logger.info(f"Wrote {path}")
    except Exception as e:
        logger.error(f"Failed to write {path}: {e}")
        raise


def append_text_file(path: str, text: str):
    ensure_dir(path)
    try:
        # Clean the content before appending
        cleaned_text = clean_yaml_content(text)
        with open(path, "a", encoding="utf-8", newline="\n") as of:
            of.write(cleaned_text)
        logger.info(f"Appended to {path}")
    except Exception as e:
        logger.error(f"Failed appending {path}: {e}")
        raise


def generate_from_template(template_path, output_dir, vars_map, suffix):
    logger.info(f"Generating files from template: {template_path}")
    template_text = read_template(template_path)
    ensure_dir(output_dir)
    for key, context in vars_map.items():
        rendered = substitute_tags(template_text, context).strip() + "\n"
        out_name = resolve_output_name(context, key, suffix)
        out_path = os.path.join(output_dir, out_name)
        write_text_file(out_path, rendered)


def generate_aggregated(template_path, output_file, vars_map):
    logger.info(f"Generating aggregated from {template_path} -> {output_file}")
    template_text = read_template(template_path)
    pieces = []
    for _, context in vars_map.items():
        pieces.append(substitute_tags(template_text, context).strip())
    result = "\n".join(pieces).strip() + "\n"
    write_text_file(output_file, result)


def build_master_context(vars_map):
    """Build dynamic context for master climate."""
    sensors_exprs = [
        f"states('sensor.stox_fancoil_{ctx['id']}') | float(0)"
        for ctx in vars_map.values()
    ]
    sensors_list = ", ".join(sensors_exprs) if sensors_exprs else ""

    # Build rooms array for master climate
    rooms_array = []
    climate_entities_list = []
    room_sensor_ids = []  # <-- NEW
    for ctx in vars_map.values():
        room_id = ctx["id"]
        rooms_array.append(
            f"{{'id': '{room_id}', "
            f"'climate': 'climate.controllo_temperatura_{room_id}', "
            f"'follow': 'input_boolean.fancoil_{room_id}_hvac_mode_follow_master'}}"
        )
        climate_entities_list.append(f"'climate.controllo_temperatura_{room_id}'")
        room_sensor_ids.append(f"'sensor.stox_fancoil_{room_id}'")  # <-- NEW

    rooms_array_str = "[" + ", ".join(rooms_array) + "]"
    climate_entities_list_str = "[" + ", ".join(climate_entities_list) + "]"
    room_sensors_list_str = "[" + ", ".join(room_sensor_ids) + "]"  # <-- NEW

    return {
        "sensors_list": sensors_list,
        "rooms_array": rooms_array_str,
        "climate_entities_list": climate_entities_list_str,
        "room_sensors_list": room_sensors_list_str,  # <-- NEW
    }


def generate_master_and_rooms(
    master_template_path, room_template_path, output_file, vars_map, extra_context=None
):
    """Write master block then append every room block."""
    logger.info(f"Generating combined master+rooms -> {output_file}")
    master_text = read_template(master_template_path)
    if extra_context:
        master_rendered = substitute_tags(master_text, extra_context).strip()
    else:
        master_rendered = master_text.strip()
    write_text_file(output_file, master_rendered + "\n")

    room_text = read_template(room_template_path)
    for _, context in vars_map.items():
        append_text_file(
            output_file, substitute_tags(room_text, context).strip() + "\n"
        )

def generate_dashboard(vars_map):
    """Generates the full dashboard YAML."""
    logger.info("Generating dashboard")
    
    main_tpl = read_template("yaml_generator/templates/dashboard_main.j2")
    termostati_tpl = read_template("yaml_generator/templates/dashboard_termostati_card.j2")
    impostazioni_tpl = read_template("yaml_generator/templates/dashboard_impostazioni_card.j2")

    termostati_cards = []
    impostazioni_cards = []

    for _, context in vars_map.items():
        # Ensure friendly_name is available
        local_ctx = dict(context)
        if "friendly_name" not in local_ctx:
            local_ctx["friendly_name"] = local_ctx.get("id", "").replace("_", " ").title()

        termostati_cards.append(substitute_tags(termostati_tpl, local_ctx))
        impostazioni_cards.append(substitute_tags(impostazioni_tpl, local_ctx))

    # Join the cards with the correct indentation
    termostati_section = "\n".join(termostati_cards)
    impostazioni_section = "\n".join(impostazioni_cards)
    
    # Substitute the placeholders in the main template
    final_dashboard = main_tpl.replace("%%TERMOSTATI_CARDS%%", termostati_section)
    final_dashboard = final_dashboard.replace("%%IMPOSTAZIONI_CARDS%%", impostazioni_section)

    write_text_file(DASHBOARD_OUTPUT_FILE, final_dashboard)
    logger.info(f"Dashboard written to {DASHBOARD_OUTPUT_FILE}")

        MASTER_INPUT_NUMBER_TEMPLATE_PATH,

        extra_context=None,

        MASTER_CLIMATE_OUTPUT_FILE,

# Pyscript service for logging and state persistence
import os
import json
from datetime import datetime

LOG_DIR = "/config/logs/fancoil_events"
LOG_FILE = f"{LOG_DIR}/fancoil_events.log"
STATE_FILE = f"{LOG_DIR}/fancoil_states.json"

def sync_archive_log():
    try:
        if os.path.exists(LOG_FILE):
            mtime = datetime.fromtimestamp(os.path.getmtime(LOG_FILE))
            ts_str = mtime.strftime("%Y-%m-%d_%H%M%S")
            new_name = f"{LOG_DIR}/fancoil_events_{ts_str}.log"
            os.rename(LOG_FILE, new_name)
    except Exception as e:
        print(f"Archive error: {e}")

def sync_write_log(line):
    try:
        if not os.path.exists(LOG_DIR):
            os.makedirs(LOG_DIR)
        with open(LOG_FILE, "a", encoding="utf-8") as f:
            f.write(line)
    except Exception as e:
        print(f"Write log error: {e}")

def sync_save_state(entity_id, value):
    try:
        if not os.path.exists(LOG_DIR):
            os.makedirs(LOG_DIR)
        states_data = {}
        if os.path.exists(STATE_FILE):
            with open(STATE_FILE, "r", encoding="utf-8") as f:
                states_data = json.load(f)
        states_data[entity_id] = value
        with open(STATE_FILE, "w", encoding="utf-8") as f:
            json.dump(states_data, f, indent=4)
    except Exception as e:
        print(f"Save state error: {e}")

def sync_read_states():
    try:
        if os.path.exists(STATE_FILE):
            with open(STATE_FILE, "r", encoding="utf-8") as f:
                return json.load(f)
    except Exception as e:
        print(f"Read states error: {e}")
    return {{}}

@service
def fancoil_events_write(message="", level="INFO"):
    ts = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    line = f"{ts} [{level}] {message}\n"
    task.executor(sync_write_log, line)

@service
def fancoil_state_save(entity_id=None, value=None):
    if not entity_id: return
    task.executor(sync_save_state, entity_id, value)

@service
def fancoil_states_restore():
    states_data = task.executor(sync_read_states)
    if not states_data: return
    for entity_id, value in states_data.items():
        domain = entity_id.split(".")[0]
        try:
            if domain == "input_number":
                input_number.set_value(entity_id=entity_id, value=float(value))
            elif domain == "input_boolean":
                if value in ["on", "True", True]:
                    input_boolean.turn_on(entity_id=entity_id)
                else:
                    input_boolean.turn_off(entity_id=entity_id)
        except Exception: pass
    fancoil_events_write("Ripristino stati da file completato", "SYSTEM")

# Trigger di inizializzazione nativo di Pyscript
@time_trigger("startup")
def startup_logic():
    task.executor(sync_archive_log)
    fancoil_events_write("=== Home Assistant / Pyscript started (New Log Session) ===", "SYSTEM")

          ### Descrizione del Sistema Fancoil
          Questo sistema gestisce i fancoil dell'appartamento, controllando la velocità della ventola in base alla differenza tra la temperatura ambiente e il setpoint desiderato.

          **Funzionamento Base:**
          - La velocità della ventola (Idle, Low, Medium, High) viene regolata dinamicamente in base alla "Distanza (Diff)" dalla temperatura del setpoint.
          - **IDLE:** La ventola è spenta, ma la valvola resta aperta se necessario (es. fancoil acceso).
          - **LOW, MEDIUM, HIGH:** La ventola si attiva a diverse velocità man mano che la differenza di temperatura aumenta.

          **Parametri Chiave:**
          - **Distanza STOP:** Se la differenza di temperatura scende al di sotto di questo valore (es. temperatura quasi raggiunta), il fancoil spegne la ventola e chiude la valvola per evitare sprechi energetici (o per prevenire un raffreddamento/riscaldamento eccessivo).
          - **Distanza IDLE/LOW/MEDIUM/HIGH:** Queste soglie definiscono quando passare alla velocità successiva della ventola. Ad esempio, se la "Distanza LOW" è 1°C, la ventola si attiverà in modalità bassa se la differenza è maggiore di 1°C.
          - **Distanza RESUME:** Rappresenta l'isteresi di ripartenza. Dopo uno STOP, il fancoil riprenderà a funzionare solo se la differenza di temperatura supera questa soglia (e dopo un "Ritardo Ripresa" minimo). Questo evita cicli di accensione/spegnimento troppo frequenti (short-cycling).
          - **Ritardo Ripresa (Minuti):** Dopo uno STOP, il sistema aspetterà questo intervallo di tempo prima di poter riavviare le ventole, anche se la temperatura dovesse superare la "Distanza RESUME".
          - **Soglia Temp Circuito:** Questi valori permettono di spegnere i fancoil se la temperatura del circuito (riscaldamento/raffrescamento) non è adeguata al funzionamento. Ad esempio, se il circuito di riscaldamento è troppo freddo, non ha senso far funzionare il fancoil.

          **Meccanismo di Isteresi:**
          L'isteresi è fondamentale per evitare oscillazioni rapide e frequenti degli stati del fancoil.
          - **STOP:** Quando la temperatura ambiente si avvicina al setpoint, il fancoil entra in stato di STOP (ventola spenta, valvola chiusa).
          - **RIPRESA:** Per riavviare il fancoil, non è sufficiente che la temperatura si discosti leggermente dal setpoint; deve superare la "Distanza RESUME" e deve essere trascorso il "Ritardo Ripresa" dall'ultimo spegnimento. Questo introduce un "cuscinetto" che stabilizza il funzionamento, riduce l'usura dei componenti e migliora l'efficienza energetica.

          Il sistema è progettato per ottimizzare il comfort e il consumo energetico, riducendo i cicli brevi e mantenendo la temperatura stabile.
